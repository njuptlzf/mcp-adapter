    ```typescript
    /**
     * MatrixReporter — vitest custom Reporter that writes an agent × section
     * matrix report at the end of every test run.
     *
     * Per D-16 / D-17:
     *   - Writes tests/reports/mcp-adapter-test-report.md (human-readable)
     *   - Writes tests/reports/mcp-adapter-test-report.json (CI / dashboard sidecar)
     *
     * Trigger: Reporter.onTestRunEnd fires once after all test modules report.
     * Each TestModule's name (from describe.each interpolation) is parsed for
     * the agent id prefix and section id; counters are incremented per cell.
     *
     * Threat-model notes (T-07-12): output paths are HARDCODED (no user
     * input, no path traversal); mkdirSync with recursive: true is idempotent.
     */
    import type { Reporter, TestModule } from "vitest/reporters";
    import { writeFileSync, mkdirSync } from "node:fs";
    import { resolve, dirname } from "node:path";
    import { fileURLToPath } from "node:url";

    interface MatrixRow {
      agent: string;
      section: string;
      pass: number;
      fail: number;
      skipped: number;
    }

    const PROJECT_ROOT = resolve(dirname(fileURLToPath(import.meta.url)), "../..");
    const REPORT_DIR = resolve(PROJECT_ROOT, "tests", "reports");
    const REPORT_MD = resolve(REPORT_DIR, "mcp-adapter-test-report.md");
    const REPORT_JSON = resolve(REPORT_DIR, "mcp-adapter-test-report.json");

    function classifySection(moduleName: string | undefined): string {
      // Best-effort: map describe names to section ids used by SKILL.md
      const n = moduleName ?? "";
      if (/Capability Gate/i.test(n)) return "Gate";
      if (/MockAgentAPI server compat/i.test(n)) return "Section4";
      if (/AgentAPI contract/i.test(n)) return "Section4-contract";
      if (/interactive visualizer/i.test(n)) return "Prebuild";
      if (/proxy-modes/i.test(n)) return "Section6-proxy";
      if (/direct-tools/i.test(n)) return "Section6-directTools";
      if (/smoke|e2e-all-servers/i.test(n)) return "Section6-E2E";
      if (/qoder-adapter-integration/i.test(n)) return "Section6-QoderIntegration";
      return "Other";
    }

    function classifyAgent(moduleName: string | undefined): string {
      // Parse "AgentAPI contract — adapter: pi" → "pi"
      const n = moduleName ?? "";
      const m = n.match(/adapter:\s*(\w+)/i);
      if (m) return m[1];
      if (/Qoder/i.test(n)) return "qoder";
      if (/Pi/i.test(n)) return "pi";
      return "env";
    }

    export default class MatrixReporter implements Reporter {
      private rows = new Map<string, MatrixRow>();

      private bucketKey(agent: string, section: string): string {
        return `${agent}::${section}`;
      }

      private bump(agent: string, section: string, kind: "pass" | "fail" | "skipped"): void {
        const key = this.bucketKey(agent, section);
        let row = this.rows.get(key);
        if (!row) {
          row = { agent, section, pass: 0, fail: 0, skipped: 0 };
          this.rows.set(key, row);
        }
        row[kind] += 1;
      }

      onTestModuleEnd(testModule: TestModule): void {
        // vitest 3 TestModule has .moduleId (path) and a tree of tasks under it
        const moduleName = testModule.moduleId ?? "unknown";
        const agent = classifyAgent(moduleName);
        const section = classifySection(moduleName);

        // Iterate child tasks if available; fall back to counting module as 1 pass
        const tasks = (testModule as unknown as { children?: { tasks?: Array<{ result?: { state?: string } }> } }).children;
        const taskList = tasks?.tasks ?? [];
        if (taskList.length === 0) {
          // Module-level result: treat as a single pass/fail
          const state = (testModule as unknown as { result?: { state?: string } }).result?.state;
          if (state === "fail") this.bump(agent, section, "fail");
          else if (state === "skip") this.bump(agent, section, "skipped");
          else this.bump(agent, section, "pass");
          return;
        }
        for (const t of taskList) {
          const state = t.result?.state;
          if (state === "fail") this.bump(agent, section, "fail");
          else if (state === "skip" || state === "todo") this.bump(agent, section, "skipped");
          else this.bump(agent, section, "pass");
        }
      }

      onTestRunEnd(endReason: { reason?: string }): void {
        mkdirSync(REPORT_DIR, { recursive: true });

        const rows = [...this.rows.values()].sort((a, b) => {
          if (a.agent !== b.agent) return a.agent.localeCompare(b.agent);
          return a.section.localeCompare(b.section);
        });

        // ---- JSON sidecar ----
        const jsonPayload = {
          generatedAt: new Date().toISOString(),
          endReason: endReason.reason ?? "unknown",
          rows,
        };
        writeFileSync(REPORT_JSON, JSON.stringify(jsonPayload, null, 2), "utf-8");

        // ---- Markdown matrix ----
        const md: string[] = [];
        md.push("# mcp-adapter Test Report — agent × section matrix");
        md.push("");
        md.push(`Generated: ${jsonPayload.generatedAt}`);
        md.push(`End reason: ${jsonPayload.endReason}`);
        md.push("");
        md.push("## Summary matrix");
        md.push("");
        md.push("| Agent | Section | Pass | Fail | Skipped |");
        md.push("|-------|---------|------|------|---------|");
        for (const r of rows) {
          md.push(`| ${r.agent} | ${r.section} | ${r.pass} | ${r.fail} | ${r.skipped} |`);
        }
        md.push("");
        md.push("## Per-agent detail");
        md.push("");
        const byAgent = new Map<string, MatrixRow[]>();
        for (const r of rows) {
          const list = byAgent.get(r.agent) ?? [];
          list.push(r);
          byAgent.set(r.agent, list);
        }
        for (const [agent, agentRows] of byAgent) {
          md.push(`### ${agent}`);
          md.push("");
          for (const r of agentRows) {
            const total = r.pass + r.fail + r.skipped;
            md.push(`- **${r.section}**: ${r.pass}/${total} pass (${r.fail} fail, ${r.skipped} skipped)`);
          }
          md.push("");
        }
        writeFileSync(REPORT_MD, md.join("\n"), "utf-8");
      }
    }
    ```
